# ==============================================================================
# 1. INTERACTIVE MOUSE & KEYBOARD TAPPER (-t)
# Hardware-level Windows tapping interface for manual signal replication.
# ==============================================================================

# Launch interactive tapper (silent mode):
python Tools/Crypto/morse.py -t

# Launch interactive tapper WITH live synthesized audio beeps:
python Tools/Crypto/morse.py -t -s

# Controls Reference:
# - LEFT CLICK   (or '.' key)  = Dot (.) [Short beep if -s]
# - RIGHT CLICK  (or '-' key)  = Dash (-) [Long beep if -s]
# - SPACEBAR (1st tap)         = Commits the current letter (e.g. ... -> (S))
# - SPACEBAR (2nd tap)         = Inserts word break (/)
# - BACKSPACE                  = Undo last tapped symbol / letter
# - ENTER                      = Prints the complete decoded message
# - ESC / Ctrl+C               = Cleanly exits tapper


# ==============================================================================
# 2. REAL-TIME REPL LISTENER (-l)
# Bidirectional prompt that automatically auto-detects Morse vs. ASCII text.
# ==============================================================================

# Start the interactive listener prompt:
python Tools/Crypto/morse.py -l

# Behavior:
# - If you paste Morse code (e.g. "... --- ..."), it decodes immediately to ASCII.
# - If you type ASCII text (e.g. "HELLO CTF"), it encodes immediately to Morse.
# - Type 'exit' or 'quit' to close.


# ==============================================================================
# 3. DIRECT CLI DECODING & ENCODING (-d, -e, -f)
# ==============================================================================

# A. Decode standard Morse string:
python Tools/Crypto/morse.py -d "... --- ..."

# B. Decode CTF Flag format:
python Tools/Crypto/morse.py -d ".... .- -.-. -.- ....- --. --- ...- -.--. -.-. - ..-. ..--.- ..-. .-.. .- --. -.--.-"

# C. Decode binary Morse (0 = dot, 1 = dash):
python Tools/Crypto/morse.py -d "000 111 000"

# D. Decode spoken or written dit-dah strings:
python Tools/Crypto/morse.py -d "di-di-dit dah-dah-dah di-di-dit"

# E. Decode Morse from a text file:
python Tools/Crypto/morse.py -f challenge_morse.txt

# F. Decode file and save output to disk:
python Tools/Crypto/morse.py -f challenge_morse.txt -o decoded_flag.txt

# G. Encode plaintext to Morse:
python Tools/Crypto/morse.py -e "HACK4GOV{CTF_FLAG}"

# H. Encode plaintext and play audible audio beeps (Windows):
python Tools/Crypto/morse.py -e "SOS" -s


# ==============================================================================
# 4. AUDIO SIGNAL DECODER FOR .WAV FILES (-a)
# Automatically extracts Morse tone pulses from CTF audio challenges.
# Reads PCM amplitude, segments on/off intervals, and outputs decoded ASCII.
# ==============================================================================

# A. Standard automatic pulse detection:
python Tools/Crypto/morse.py -a challenge_audio.wav

# B. Audio with background noise (increase energy threshold, default: 0.15):
python Tools/Crypto/morse.py -a noisy_audio.wav --threshold 0.25

# C. Faint or quiet audio recording (lower energy threshold):
python Tools/Crypto/morse.py -a quiet_audio.wav --threshold 0.08


# ==============================================================================
# 5. LIVE MICROPHONE AUDIO LISTENER (--mic)
# Captures sound directly from PC microphone, performs FFT spectral isolation,
# suppresses human speech / room noise (<460 Hz), and prints decoded letters live.
# ==============================================================================

# A. Start live microphone listener (Auto-calibrates noise floor & 520-1250 Hz beeps):
python Tools/Crypto/morse.py --mic

# B. Target a known specific frequency tone (e.g. 800 Hz):
python Tools/Crypto/morse.py --mic --freq 800 --threshold 0.02

# C. Stream live transcribed text to file:
python Tools/Crypto/morse.py --mic -o live_transcript.txt
