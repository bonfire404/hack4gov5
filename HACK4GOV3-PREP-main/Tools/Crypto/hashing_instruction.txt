# ==============================================================================
# 3. UNKNOWN HASH FORMAT IDENTIFIER
# Analyzes digest bit lengths, hex encoding, and modular crypt format prefixes ($...).
# ==============================================================================

# 32 characters (128-bit) -> Identifies MD5, NTLM:
python Tools/Crypto/hasher.py -i "5d41402abc4b2a76b9719d911017c592"

# 40 characters (160-bit) -> Identifies SHA-1, RIPEMD-160:
python Tools/Crypto/hasher.py -i "2fd4e1c67a2d28fced849ee1bb76e7391b93eb12"

# 64 characters (256-bit) -> Identifies SHA-256:
python Tools/Crypto/hasher.py -i "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"

# Modular Crypt & Password Hash Prefixes:
# - $1$  -> Linux MD5 shadow hash
# - $2a$, $2b$, $2y$ -> Bcrypt
# - $5$  -> SHA-256 shadow hash
# - $6$  -> SHA-512 shadow hash
# - $argon2id$ -> Argon2
python Tools/Crypto/hasher.py -i "$2a$10$N9qo8uLOickgx2ZMRZoMyeIjZAgcfl7p92ldGxad68LJZdL17lhWy"
python Tools/Crypto/hasher.py -i "$6$rounds=5000$saltsalt$5.jWv2K4kG..."
